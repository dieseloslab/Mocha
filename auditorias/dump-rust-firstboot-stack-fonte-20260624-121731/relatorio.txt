   129	    log.line(format!("+ {} {}", program, args.join(" ")));
   130	
   131	    let output = Command::new(program)
   132	        .env("LC_ALL", "C")

   134	        .args(args)
   135	        .output()
   136	        .map_err(|e| format!("falha ao executar {}: {}", program, e))?;
   137	
   138	    let stdout = String::from_utf8_lossy(&output.stdout).to_string();
   139	    let stderr = String::from_utf8_lossy(&output.stderr).to_string();
   140	
   141	    if !stdout.is_empty() {
   142	        log.raw(&stdout);
   143	    }
   144	
   145	    if !stderr.is_empty() {
   146	        log.raw(&stderr);
   147	    }
   148	
   149	    if output.status.success() {
   150	        Ok(stdout)
   151	    } else {
   152	        Err(format!(
   153	            "comando falhou: {} {}\nstderr: {}",
   154	            program,
   155	            args.join(" "),
   156	            stderr.trim()
   157	        ))
   158	    }
   159	}

   248	fn strip_cachyos_blocks(conf: &str) -> String {
   249	    let mut out = String::new();
   250	    let mut skip = false;
   251	
   252	    for line in conf.lines() {
   253	        let trimmed = line.trim();
   254	
   255	        if trimmed.starts_with("[cachyos") && trimmed.ends_with(']') {
   256	            skip = true;
   257	            continue;
   258	        }
   259	
   260	        if trimmed.starts_with('[') && trimmed.ends_with(']') {
   261	            skip = false;
   262	        }
   263	
   264	        if !skip {
   265	            out.push_str(line);
   266	            out.push('\n');
   267	        }
   268	    }
   269	
   270	    out
   271	}

   273	fn replace_architecture(conf: &str, arch_line: &str) -> Result<String, String> {
   274	    if !conf.lines().any(|l| l.trim() == "[options]") {
   275	        return Err("pacman.conf sem bloco [options]".to_string());
   276	    }
   277	
   278	    let mut out = String::new();
   279	    let mut in_options = false;
   280	    let mut inserted = false;
   281	
   282	    for line in conf.lines() {
   283	        let trimmed = line.trim();
   284	
   285	        if trimmed == "[options]" {
   286	            in_options = true;
   287	            out.push_str(line);
   288	            out.push('\n');
   289	            continue;
   290	        }
   291	
   292	        if in_options && trimmed.starts_with('[') && trimmed.ends_with(']') {
   293	            if !inserted {
   294	                out.push_str(arch_line);
   295	                out.push('\n');
   296	                inserted = true;
   297	            }
   298	            in_options = false;
   299	            out.push_str(line);
   300	            out.push('\n');
   301	            continue;
   302	        }
   303	
   304	        if in_options && trimmed.starts_with("Architecture") {
   305	            if !inserted {
   306	                out.push_str(arch_line);
   307	                out.push('\n');
   308	                inserted = true;
   309	            }
   310	            continue;
   311	        }
   312	
   313	        out.push_str(line);
   314	        out.push('\n');
   315	    }
   316	
   317	    if in_options && !inserted {
   318	        out.push_str(arch_line);
   319	        out.push('\n');
   320	        inserted = true;
   321	    }
   322	
   323	    if !inserted {
   324	        return Err("não consegui inserir Architecture no pacman.conf temporário".to_string());
   325	    }
   326	
   327	    Ok(out)
   328	}

   330	fn write_temp_pacman_conf(
   331	    log: &mut Logger,
   332	    profile: &CpuProfile,
   333	    work: &Path,
   334	) -> Result<PathBuf, String> {
   335	    let original = fs::read_to_string("/etc/pacman.conf")
   336	        .map_err(|e| format!("não consegui ler /etc/pacman.conf: {}", e))?;
   337	
   338	    let sanitized = strip_cachyos_blocks(&original);
   339	    let base = replace_architecture(&sanitized, &profile.arch_line)?;
   340	
   341	    let conf_path = work.join("pacman-cachyos-temporario.conf");
   342	    let mirror_opt = work.join("mirrorlist-cachyos-otimizado");
   343	    let mirror_x64 = work.join("mirrorlist-cachyos-x86_64");
   344	
   345	    let opt_servers = format!(
   346	        "Server = https://mirror.cachyos.org/repo/{}/$repo\nServer = https://cdn.cachyos.org/repo/{}/$repo\nServer = https://cdn77.cachyos.org/repo/{}/$repo\n",
   347	        profile.mirror_arch, profile.mirror_arch, profile.mirror_arch
   348	    );
   349	
   350	    let x64_servers = "Server = https://mirror.cachyos.org/repo/x86_64/$repo\nServer = https://cdn.cachyos.org/repo/x86_64/$repo\nServer = https://cdn77.cachyos.org/repo/x86_64/$repo\n";
   351	
   352	    fs::write(&mirror_opt, opt_servers)
   353	        .map_err(|e| format!("não consegui escrever mirrorlist otimizada: {}", e))?;
   354	
   355	    fs::write(&mirror_x64, x64_servers)
   356	        .map_err(|e| format!("não consegui escrever mirrorlist x86_64: {}", e))?;
   357	
   358	    let mut temp = base;
   359	    temp.push_str("\n# Mocha temporário — CachyOS restrito; não copiar para /etc/pacman.conf\n");
   360	
   361	    for repo in &profile.optimized_repos {
   362	        temp.push_str(&format!("[{}]\nInclude = {}\n", repo, mirror_opt.display()));
   363	    }
   364	
   365	    temp.push_str(&format!("[cachyos]\nInclude = {}\n", mirror_x64.display()));
   366	
   367	    fs::write(&conf_path, temp)
   368	        .map_err(|e| format!("não consegui escrever pacman.conf temporário: {}", e))?;
   369	
   370	    log.line(format!("Arquitetura temporária: {}", profile.arch_line));
   371	    log.line(format!("Arquivo temporário: {}", conf_path.display()));
   372	
   373	    Ok(conf_path)
   374	}

   379

   489	fn build_plan(
   490	    log: &mut Logger,
   491	    conf: &Path,
   492	    profile: &CpuProfile,
   493	    nvidia: bool,
   494	) -> Result<Vec<PackageTarget>, String> {
   495	    let mut repos = profile.optimized_repos.clone();
   496	    repos.push("cachyos".to_string());
   497	
   498	    let kernel = find_package(log, conf, &repos, "linux-cachyos")?;
   499	    let headers = find_package(log, conf, &repos, "linux-cachyos-headers")?;
   500	
   501	    validate_arch(&kernel, profile)?;
   502	    validate_arch(&headers, profile)?;
   503	
   504	    if kernel.version != headers.version {
   505	        return Err(format!(
   506	            "kernel e headers não estão casados: linux-cachyos={} headers={}",
   507	            kernel.version, headers.version
   508	        ));
   509	    }
   510	
   511	    let mut targets = vec![kernel.clone(), headers];
   512	
   513	    if nvidia {
   514	        let module = if package_exists(log, conf, &kernel.repo, "linux-cachyos-nvidia-open") {
   515	            find_package(
   516	                log,
   517	                conf,
   518	                &[kernel.repo.clone()],
   519	                "linux-cachyos-nvidia-open",
   520	            )?
   521	        } else if package_exists(log, conf, &kernel.repo, "linux-cachyos-nvidia") {
   522	            find_package(log, conf, &[kernel.repo.clone()], "linux-cachyos-nvidia")?
   523	        } else {
   524	            find_package(log, conf, &repos, "linux-cachyos-nvidia-open")
   525	                .or_else(|_| find_package(log, conf, &repos, "linux-cachyos-nvidia"))?
   526	        };
   527	
   528	        validate_arch(&module, profile)?;
   529	
   530	        if module.version != kernel.version {
   531	            return Err(format!(
   532	                "módulo NVIDIA do kernel não casa com o kernel: linux-cachyos={} módulo={}",
   533	                kernel.version, module.version
   534	            ));
   535	        }
   536	
   537	        targets.push(module);
   538	
   539	        let x64_repos = vec![
   540	            "cachyos".to_string(),
   541	            "extra".to_string(),
   542	            "multilib".to_string(),
   543	        ];
   544	
   545	        for pkg in ["nvidia-utils", "lib32-nvidia-utils", "opencl-nvidia"] {
   546	            let t = find_package(log, conf, &x64_repos, pkg)?;
   547	            validate_arch(&t, profile)?;
   548	            targets.push(t);
   549	        }
   550	    }
   551	
   552	    Ok(targets)
   553	}

   489	fn build_plan(
   490	    log: &mut Logger,
   491	    conf: &Path,
   492	    profile: &CpuProfile,
   493	    nvidia: bool,
   494	) -> Result<Vec<PackageTarget>, String> {
   495	    let mut repos = profile.optimized_repos.clone();
   496	    repos.push("cachyos".to_string());
   497	
   498	    let kernel = find_package(log, conf, &repos, "linux-cachyos")?;
   499	    let headers = find_package(log, conf, &repos, "linux-cachyos-headers")?;
   500	
   501	    validate_arch(&kernel, profile)?;
   502	    validate_arch(&headers, profile)?;
   503	
   504	    if kernel.version != headers.version {
   505	        return Err(format!(
   506	            "kernel e headers não estão casados: linux-cachyos={} headers={}",
   507	            kernel.version, headers.version
   508	        ));
   509	    }
   510	
   511	    let mut targets = vec![kernel.clone(), headers];
   512	
   513	    if nvidia {
   514	        let module = if package_exists(log, conf, &kernel.repo, "linux-cachyos-nvidia-open") {
   515	            find_package(
   516	                log,
   517	                conf,
   518	                &[kernel.repo.clone()],
   519	                "linux-cachyos-nvidia-open",
   520	            )?
   521	        } else if package_exists(log, conf, &kernel.repo, "linux-cachyos-nvidia") {
   522	            find_package(log, conf, &[kernel.repo.clone()], "linux-cachyos-nvidia")?
   523	        } else {
   524	            find_package(log, conf, &repos, "linux-cachyos-nvidia-open")
   525	                .or_else(|_| find_package(log, conf, &repos, "linux-cachyos-nvidia"))?
   526	        };
   527	
   528	        validate_arch(&module, profile)?;
   529	
   530	        if module.version != kernel.version {
   531	            return Err(format!(
   532	                "módulo NVIDIA do kernel não casa com o kernel: linux-cachyos={} módulo={}",
   533	                kernel.version, module.version
   534	            ));
   535	        }
   536	
   537	        targets.push(module);
   538	
   539	        let x64_repos = vec![
   540	            "cachyos".to_string(),
   541	            "extra".to_string(),
   542	            "multilib".to_string(),
   543	        ];

   545	        for pkg in ["nvidia-utils", "lib32-nvidia-utils", "opencl-nvidia"] {
   546	            let t = find_package(log, conf, &x64_repos, pkg)?;
   547	            validate_arch(&t, profile)?;
   548	            targets.push(t);
   549	        }
   550	    }
   551	
   552	    Ok(targets)
   553	}
